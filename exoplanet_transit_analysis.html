# Capped, not full-mission: keeps quarter-boundary artifacts limited and
# keeps periodogram grids tractable.
N_QUARTERS_CAP = 6

# Narrowed back down: stay below ~90-day quarter-boundary systematics,
# and matches where the majority of the confirmed sample's periods live.
PERIOD_RANGE = (0.5, 50.0)
MAX_GRID_POINTS = 200_000

def choose_frequency_factor(time_days, duration_grid, min_period, max_period):
    dummy_flux = np.ones_like(time_days)
    bls = BoxLeastSquares(time_days, dummy_flux)
    factor = 1.0
    while True:
        grid = bls.autoperiod(
            duration_grid, minimum_period = min_period,
            maximum_period = max_period, frequency_factor = factor,
        )
        if len(grid) <= MAX_GRID_POINTS:
            return factor, len(grid)
        factor *= 2.0

def compute_sde(pg):
    """Signal Detection Efficiency: how far the peak power stands out
    from the periodogram's own noise level."""
    power = pg.power.value
    peak = power.max()
    mean = np.mean(power)
    std = np.std(power)
    if std == 0:
        return 0.0
    return (peak - mean) / std

def extract_features(target):
    try:
        search = lk.search_lightcurve(target, author="Kepler", cadence="long")
        if len(search) == 0:
            print(f"No data found for {target}")
            return None

        # FIX: cap quarters deliberately, don't slice-then-download-all,
        # and don't take the full mission (quarter-boundary artifacts).
        lc = search[:N_QUARTERS_CAP].download_all().stitch()

        lc_clean = (
            lc.remove_nans()
              .remove_outliers()
              .flatten(window_length=401)
              .normalize()
        )

        n_cadences = len(lc_clean.flux)
        baseline_days = float((lc_clean.time.max() - lc_clean.time.min()).value)

        duration_grid = np.linspace(0.05, 0.30, 5)
        min_period, max_period = PERIOD_RANGE
        # Don't search periods that can't have 2 transits in the baseline
        max_period = min(max_period, baseline_days / 2)
        if max_period <= min_period:
            print(f"Baseline too short for period search: {target}")
            return None

        time_days = lc_clean.time.value
        freq_factor, n_grid_points = choose_frequency_factor(
            time_days, duration_grid, min_period, max_period
        )

        pg = lc_clean.to_periodogram(
            method = "bls",
            minimum_period = min_period,
            maximum_period = max_period,
            duration = duration_grid,
            frequency_factor = freq_factor,
        )
        sde = compute_sde(pg)
        SDE_THRESHOLD = 6.0  # commonly used cutoff in the transit-search literature
        
        significant = sde >= SDE_THRESHOLD
        
        best_period = float(pg.period_at_max_power.value)
        depth = float(pg.depth_at_max_power.value)
        duration = float(pg.duration_at_max_power.value)
        power = float(pg.max_power.value)
        noise = float(np.std(lc_clean.flux.value))
        if noise <= 0:
            return None

        # FIX: reject detections with fewer than 2 observable transits --
        # not statistically constrained by the data.
        n_transits = baseline_days / best_period
        if n_transits < 2:
            print(f"Fewer than 2 transits for {target} (period={best_period:.1f}d); skipping")
            return None

        return {
            "target": target,
            "period": best_period,
            "depth": depth,
            "duration": duration,
            "power": power,
            "depth_to_noise": depth / noise,
            "n_cadences": n_cadences,
            "baseline_days": baseline_days,
            "frequency_factor_used": freq_factor,
            "n_grid_points": n_grid_points,
            "sde": sde,
            "significant_detection": significant,

        }

    except Exception as e:
        print(f"Failed for {target}: {e}")
        return None

CHECKPOINT_PATH = "features_checkpoint.csv"

def extract_features_with_retry(target, max_retries=3, backoff_seconds=10):
    for attempt in range(1, max_retries + 1):
        result = extract_features(target)
        if result is not None:
            return result
        print(f"  retry {attempt}/{max_retries} for {target} in {backoff_seconds}s...")
        time.sleep(backoff_seconds)
    return None

rows = []
already_done = set()
if os.path.exists(CHECKPOINT_PATH):
    prev = pd.read_csv(CHECKPOINT_PATH)
    rows = prev.to_dict("records")
    already_done = set(prev["target"])
    print(f"Resuming: {len(already_done)} targets already processed")

def process(kepid, label, i, total, tag):
    target = f"KIC {int(kepid)}"
    if target in already_done:
        return
    print(f"[{tag} {i + 1}/{total}] {target}")
    result = extract_features_with_retry(target)
    if result is not None:
        result["label"] = label
        rows.append(result)
        pd.DataFrame(rows).to_csv(CHECKPOINT_PATH, index=False)

def closest_known_period(bls_period, known_periods, max_harmonic=4):
    known_periods = np.asarray(known_periods, dtype=float)
    candidates = list(known_periods)
    for n in range(2, max_harmonic + 1):
        candidates.extend(known_periods * n)
        candidates.extend(known_periods / n)
    candidates = np.array(candidates)
    rel_err = np.abs(bls_period - candidates) / candidates
    best_idx = np.argmin(rel_err)
    return float(candidates[best_idx]), float(rel_err[best_idx])

from sklearn.model_selection import train_test_split, StratifiedKFold, cross_val_score, permutation_test_score
from sklearn.ensemble import RandomForestClassifier
from sklearn.inspection import permutation_importance
from sklearn.metrics import classification_report

df_real["log_power"] = np.log10(df_real["power"].clip(lower=1e-10))

FEATURE_SETS = {
    "without_period_or_power": ["depth", "duration", "depth_to_noise"],
    "without_period": ["depth", "duration", "log_power", "depth_to_noise"],
    "with_period":    ["period", "depth", "duration", "log_power", "depth_to_noise"],
}

cv = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)

for name, feature_cols in FEATURE_SETS.items():
    print(f"\n--- {name} ---")
    df_model = df_real.dropna(subset=feature_cols).copy()
    X = df_model[feature_cols]
    y = df_model["label"]

    X_train, X_test, y_train, y_test = train_test_split(
        X, y, test_size = 0.30, random_state = 42, stratify = y
    )

    model = RandomForestClassifier(n_estimators = 100, random_state = 42, n_jobs = -1)
    model.fit(X_train, y_train)
    pred = model.predict(X_test)

    print(classification_report(y_test, pred, zero_division=0))

    perm_result = permutation_importance(
        model, X_test, y_test, n_repeats = 30, random_state = 42, n_jobs = -1
    )
    print("Permutation importance (unbiased):")
    for f, m, s in zip(feature_cols, perm_result.importances_mean, perm_result.importances_std):
        print(f"  {f}: {m:.3f} +/- {s:.3f}")

    scores = cross_val_score(model, X_train, y_train, cv = cv, n_jobs = -1)
    print(f"5-fold CV accuracy: {scores.mean():.3f} +/- {scores.std():.3f}")

    perm_model = RandomForestClassifier(n_estimators = 100, random_state = 42, n_jobs = 1)
    obs_score, _, pvalue = permutation_test_score(
        perm_model, X_train, y_train, scoring="accuracy",
        cv = cv, n_permutations = 1000, random_state = 42, n_jobs = -1,
    )
    print(f"Permutation test -- observed accuracy: {obs_score:.3f}, p-value: {pvalue:.3f}")
